Tooltips are not a separate `@viraui/charts` component — you compose `useTooltip` / `TooltipWithBounds` with your own body. These examples show the patterns Storybook teaches for stacked activity bars.

## Common examples

### Line indicator

Tall color swatch beside each stacked row.

<Example preview={<ChartsTooltipIndicatorLineDemo />}>

```tsx
import {
  AxisBottom,
  Bar,
  BarRounded,
  Group,
  TooltipWithBounds,
  scaleBand,
  scaleLinear,
  useTooltip,
} from '@viraui/charts';

const { tooltipData, tooltipLeft, tooltipTop, tooltipOpen, showTooltip, hideTooltip } =
  useTooltip<ActivityPoint>();

const xScale = scaleBand({ domain: days, range: [0, innerWidth], padding: 0.3 });
const yScale = scaleLinear({ domain: [0, yMax], range: [innerHeight, 0], nice: true });

<svg width={width} height={height} onMouseLeave={hideTooltip}>
  <Group left={margin.left} top={margin.top}>
    {activityData.map((d) => {
      const barX = xScale(d.day) ?? 0;
      const runningY = yScale(d.running) ?? 0;
      const totalY = yScale(d.running + d.swimming) ?? 0;
      return (
        <Group
          key={d.day}
          onMouseMove={() =>
            showTooltip({
              tooltipData: d,
              tooltipLeft: margin.left + barX + xScale.bandwidth() / 2,
              tooltipTop: margin.top + totalY,
            })
          }
        >
          <Bar x={barX} y={runningY} width={xScale.bandwidth()} height={innerHeight - runningY} />
          <BarRounded
            x={barX}
            y={totalY}
            width={xScale.bandwidth()}
            height={runningY - totalY}
            all={false}
            top
          />
        </Group>
      );
    })}
    <AxisBottom top={innerHeight} scale={xScale} hideAxisLine hideTicks />
  </Group>
</svg>
{tooltipOpen && tooltipData ? (
  <TooltipWithBounds top={tooltipTop} left={tooltipLeft}>
    {/* title + tall swatch rows for running / swimming */}
  </TooltipWithBounds>
) : null}
```

</Example>

### No indicator

Title and values only — no swatch column.

<Example preview={<ChartsTooltipIndicatorNoneDemo />}>

```tsx
import { TooltipWithBounds, defaultStyles, useTooltip } from '@viraui/charts';

{tooltipOpen && tooltipData ? (
  <TooltipWithBounds top={tooltipTop} left={tooltipLeft} style={defaultStyles}>
    {tooltipData.title ? <div>{tooltipData.title}</div> : null}
    {tooltipData.rows.map((row) => (
      <div key={row.label} style={{ display: 'flex', justifyContent: 'space-between', gap: 12 }}>
        <span>{row.label}</span>
        <strong>{row.value}</strong>
      </div>
    ))}
  </TooltipWithBounds>
) : null}
```

</Example>

### Custom title

Fixed title row instead of the raw day key.

<Example preview={<ChartsTooltipLabelCustomDemo />}>

```tsx
type TitleFor = (d: ActivityPoint) => string | undefined;

const customLabelTitle: TitleFor = () => 'Activities';

showTooltip({
  tooltipData: {
    title: customLabelTitle(datum),
    rows: [
      { label: 'Running', value: datum.running, color: 'var(--highlight-blue)' },
      { label: 'Swimming', value: datum.swimming, color: 'var(--highlight-acid)' },
    ],
  },
  tooltipLeft,
  tooltipTop,
});
```

</Example>

### Long date title

`Intl.DateTimeFormat` for a readable date in the title row.

<Example preview={<ChartsTooltipLabelLongDateDemo />}>

```tsx
type TitleFor = (d: ActivityPoint) => string | undefined;

const longDateTitle: TitleFor = (d) =>
  new Intl.DateTimeFormat('en-US', { dateStyle: 'medium' }).format(new Date(d.day));

showTooltip({
  tooltipData: {
    title: longDateTitle(datum),
    rows: [
      { label: 'Running', value: datum.running, color: 'var(--highlight-blue)' },
      { label: 'Swimming', value: datum.swimming, color: 'var(--highlight-acid)' },
    ],
  },
  tooltipLeft,
  tooltipTop,
});
```

</Example>

### Formatter

Append units or custom value nodes in the body.

<Example preview={<ChartsTooltipFormatterDemo />}>

```tsx
const kcalValue = (n: number) => (
  <>
    {n}
    <span style={{ color: 'var(--global-muted)' }}> kcal</span>
  </>
);

{rows.map((row) => (
  <div key={row.label}>
    <span>{row.label}</span>
    <strong>{formatValue ? formatValue(row.value) : row.value}</strong>
  </div>
))}
```

</Example>

### Icons

Series icons instead of color swatches when the metaphor is activity type.

<Example preview={<ChartsTooltipIconsDemo />}>

```tsx
import { PersonSimpleWalk, Waves } from '@phosphor-icons/react';

const seriesMeta = {
  running: { label: 'Running', color: 'var(--highlight-blue)', icon: <PersonSimpleWalk size={16} /> },
  swimming: { label: 'Swimming', color: 'var(--highlight-acid)', icon: <Waves size={16} /> },
};

{/* render meta.icon instead of a color swatch when indicator="icon" */}
```

</Example>

### Advanced

Combine indicator, formatter, and a total row.

<Example preview={<ChartsTooltipAdvancedDemo />}>

```tsx
const total = rows.reduce((sum, row) => sum + row.value, 0);

<TooltipWithBounds top={tooltipTop} left={tooltipLeft} style={cardStyle}>
  {title ? <div>{title}</div> : null}
  {rows.map((row) => (
    <div key={row.label}>
      <span style={{ background: row.color /* or icon */ }} />
      <span>{row.label}</span>
      <strong>{formatValue?.(row.value) ?? row.value}</strong>
    </div>
  ))}
  {showTotal ? (
    <div>
      <span>Total</span>
      <strong>{formatValue?.(total) ?? total}</strong>
    </div>
  ) : null}
</TooltipWithBounds>
```

</Example>
