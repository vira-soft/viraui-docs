Tooltips are not a separate `@viraui/charts` component — you compose `useTooltip` / `TooltipWithBounds` with your own body. Each example uses a different Health / Fitness host chart so the tooltip pattern stays clear without repeating the same bars.

## Common examples

### Line indicator

Weekly **Activity** stacks (Move + Exercise). Tall color swatches beside each series row.

<Example preview={<ChartsTooltipIndicatorLineDemo />}>

```tsx
import {
  AxisBottom,
  Bar,
  BarRounded,
  Group,
  TooltipWithBounds,
  scaleBand,
  scaleLinear,
  useTooltip,
} from '@viraui/charts';

const { tooltipData, tooltipLeft, tooltipTop, tooltipOpen, showTooltip, hideTooltip } =
  useTooltip<ActivityPoint>();

const xScale = scaleBand({ domain: days, range: [0, innerWidth], padding: 0.3 });
const yScale = scaleLinear({ domain: [0, yMax], range: [innerHeight, 0], nice: true });

<svg width={width} height={height} onMouseLeave={hideTooltip}>
  <Group left={margin.left} top={margin.top}>
    {activityData.map((d) => {
      const barX = xScale(d.day) ?? 0;
      const moveY = yScale(d.move) ?? 0;
      const totalY = yScale(d.move + d.exercise) ?? 0;
      return (
        <Group
          key={d.day}
          onMouseMove={() =>
            showTooltip({
              tooltipData: d,
              tooltipLeft: margin.left + barX + xScale.bandwidth() / 2,
              tooltipTop: margin.top + totalY,
            })
          }
        >
          <Bar x={barX} y={moveY} width={xScale.bandwidth()} height={innerHeight - moveY} />
          <BarRounded
            x={barX}
            y={totalY}
            width={xScale.bandwidth()}
            height={moveY - totalY}
            all={false}
            top
          />
        </Group>
      );
    })}
    <AxisBottom top={innerHeight} scale={xScale} hideAxisLine hideTicks />
  </Group>
</svg>
{tooltipOpen && tooltipData ? (
  <TooltipWithBounds top={tooltipTop} left={tooltipLeft}>
    {/* title + tall swatch rows for Move / Exercise */}
  </TooltipWithBounds>
) : null}
```

</Example>

### No indicator

**Heart Rate** line — date title and bpm value, no swatch column.

<Example preview={<ChartsTooltipIndicatorNoneDemo />}>

```tsx
import {
  AxisBottom,
  Circle,
  Group,
  LinePath,
  TooltipWithBounds,
  curveMonotoneX,
  scaleLinear,
  scalePoint,
  useTooltip,
} from '@viraui/charts';

<svg width={width} height={height} onMouseLeave={hideTooltip} onMouseMove={onMove}>
  <Group left={margin.left} top={margin.top}>
    <LinePath
      data={heartSeries}
      x={(d) => xScale(d.key) ?? 0}
      y={(d) => yScale(d.bpm) ?? 0}
      stroke="var(--highlight-red)"
      curve={curveMonotoneX}
    />
    {tooltipData ? (
      <Circle cx={xScale(tooltipData.key)} cy={yScale(tooltipData.bpm)} r={4} fill="var(--highlight-red)" />
    ) : null}
    <AxisBottom top={innerHeight} scale={xScale} hideAxisLine hideTicks />
  </Group>
</svg>
{tooltipOpen && tooltipData ? (
  <TooltipWithBounds top={tooltipTop} left={tooltipLeft}>
    <div>{tooltipData.dateLabel}</div>
    <div>
      <span>Heart Rate</span>
      <strong>
        {tooltipData.bpm}
        <span> bpm</span>
      </strong>
    </div>
  </TooltipWithBounds>
) : null}
```

</Example>

### Custom title

**Sleep** indigo area with a fixed `Sleep` title row.

<Example preview={<ChartsTooltipLabelCustomDemo />}>

```tsx
showTooltip({
  tooltipData: {
    title: 'Sleep',
    rows: [{ label: 'Time Asleep', value: datum.hours, color: 'var(--highlight-indigo)' }],
  },
  tooltipLeft,
  tooltipTop,
});
```

</Example>

### Formatter

**Walk distance** magenta line — append a `km` unit beside the value.

<Example preview={<ChartsTooltipFormatterDemo />}>

```tsx
const kmValue = (n: number) => (
  <>
    {n.toFixed(1)}
    <span style={{ color: 'var(--global-muted)' }}> km</span>
  </>
);

{rows.map((row) => (
  <div key={row.label}>
    <span>{row.label}</span>
    <strong>{formatValue ? formatValue(row.value) : row.value}</strong>
  </div>
))}
```

</Example>

### Icons

**Workout mix** pie — series icons instead of color swatches.

<Example preview={<ChartsTooltipIconsDemo />}>

```tsx
import { Bicycle, PersonSimpleSwim, PersonSimpleWalk } from '@phosphor-icons/react';
import { Group, Pie } from '@viraui/charts';

const slices = [
  { label: 'Walk', value: 42, color: 'var(--highlight-cyan)', icon: <PersonSimpleWalk size={16} /> },
  { label: 'Cycle', value: 28, color: 'var(--highlight-acid)', icon: <Bicycle size={16} /> },
  { label: 'Swim', value: 18, color: 'var(--highlight-blue)', icon: <PersonSimpleSwim size={16} /> },
];

<Group top={centerY} left={centerX}>
  <Pie data={slices} pieValue={(d) => d.value} outerRadius={radius} padAngle={0.02}>
    {({ arcs, path }) =>
      arcs.map((arc) => (
        <path key={arc.data.label} d={path(arc) ?? undefined} fill={arc.data.color} />
      ))
    }
  </Pie>
</Group>

{/* tooltip body renders meta.icon instead of a color swatch */}
```

</Example>

### Advanced

**Macro** stacked area (protein / carbs / fat) painted with literal `--color-*-30` fills, gram formatter, and a total row.

<Example preview={<ChartsTooltipAdvancedDemo />}>

```tsx
const proteinColor = 'var(--color-red-30)';
const carbsColor = 'var(--color-cyan-30)';
const fatColor = 'var(--color-yellow-30)';

const total = rows.reduce((sum, row) => sum + row.value, 0);

<TooltipWithBounds top={tooltipTop} left={tooltipLeft} style={cardStyle}>
  {title ? <div>{title}</div> : null}
  {rows.map((row) => (
    <div key={row.label}>
      <span style={{ background: row.color }} />
      <span>{row.label}</span>
      <strong>{formatValue?.(row.value) ?? row.value}</strong>
    </div>
  ))}
  {showTotal ? (
    <div>
      <span>Total</span>
      <strong>{formatValue?.(total) ?? total}</strong>
    </div>
  ) : null}
</TooltipWithBounds>
```

</Example>
